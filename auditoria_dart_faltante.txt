class AdminNegociosRemote {
  AdminNegociosRemote([SupabaseRestClient? api])
    : _api = api ?? SupabaseRestClient();
  final SupabaseRestClient _api;

  Future<List<AdminNegocioRow>> listar({
    String busqueda = '',
    String? estado,
    String? plan,
    String orden = 'reciente',
    int limite = 20,
    int offset = 0,
  }) async {
    final res = await _api.rpc('admin_listar_negocios', {
      'p_busqueda': busqueda.trim().isEmpty ? null : busqueda.trim(),
      'p_filtro_estado': (estado == null || estado.isEmpty) ? null : estado,
      'p_filtro_plan': (plan == null || plan.isEmpty) ? null : plan,
      'p_orden': orden,
      'p_limite': limite,
      'p_desplazamiento': offset,
    });
    final lista = res is List ? res : const <dynamic>[];
    return lista
        .map(
          (e) => AdminNegocioRow.fromMap(Map<String, dynamic>.from(e as Map)),
        )
        .toList();
  }

  Future<AdminNegocioDetalle> detalle(String negocioId) async {
    final res = await _api.rpc('admin_detalle_negocio', {
      'p_negocio_id': negocioId,
    });
    if (res is! Map) throw Exception('Respuesta de detalle inválida');
    return AdminNegocioDetalle.fromJson(Map<String, dynamic>.from(res));
  }

  Future<void> _accion(String fn, Map<String, dynamic> params) =>
      _api.rpc(fn, params);

  Future<void> activar(String id, {String? motivo}) => _accion(
    'admin_activar_negocio',
    {'p_negocio_id': id, 'p_motivo': motivo},
  );

  Future<void> suspender(String id, String motivo) => _accion(
    'admin_suspender_negocio',
    {'p_negocio_id': id, 'p_motivo': motivo},
  );

  Future<void> renovar(
    String id,
    DateTime fechaFin, {
    String? plan,
    String? vigenciaTipo,
    String? motivo,
  }) => _accion('admin_renovar_negocio', {
    'p_negocio_id': id,
    'p_fecha_fin': fechaFin.toUtc().toIso8601String(),
    'p_plan': plan,
    'p_vigencia_tipo': vigenciaTipo,
    'p_motivo': motivo,
  });

  Future<void> registrarConcesion({
    required String id,
    required String tipo,
    required DateTime inicio,
    required DateTime fin,
    String origen = 'manual',
    String? motivo,
  }) => _accion('admin_registrar_concesion', {
    'p_negocio_id': id,
    'p_tipo': tipo,
    'p_fecha_inicio': inicio.toIso8601String().substring(0, 10),
    'p_fecha_fin': fin.toIso8601String().substring(0, 10),
    'p_origen': origen,
    'p_motivo': motivo,

  Future<LicenciaInfo?> ensureLicenciaInicial() async {
    final rows = await _db.select(_db.licenciasTable).get();
    if (rows.isNotEmpty) {
      rows.sort((a, b) => b.id.compareTo(a.id));
      final dev = rows.where((r) => r.plan == 'desarrollador' || r.vitalicia);
      final row = dev.isNotEmpty ? dev.first : rows.first;
      return _mapYActualizarEstado(row);

    // Equipo vacío pero negocio ya vinculado (reinstalación): restaurar concesión.

          final ok = await LicenciaSync(_db).pullDesdeNegocio();

            if (after != null) return after;
          }
        } catch (e) {
          // ignore: avoid_print
          print('ensureLicenciaInicial pull: $e');
        }
      }
    }

    // Sin contrato remoto todavía no se concede la prueba. El flujo de alta
    // deberá validar este equipo antes de crear una licencia local.
    return null;

import 'package:flutter_secure_storage/flutter_secure_storage.dart';

/// JWT de la nube. Si no hay token vigente, el REST usa la anon key.
/// No bloquea el uso offline de la app: solo afecta sync/API remota.
class SesionNube {
  static const _kToken = 'supabase_access_jwt_v2';
  static const _kExp = 'supabase_access_exp_v2';
  static const _kRefresh = 'supabase_refresh_jwt_v2';

  static const _secure = FlutterSecureStorage();

  static Future<void> guardar({
    required String token,
    required int expiresInSec,
    String? refreshToken,
  }) async {
    await _secure.write(key: _kToken, value: token);
    final exp = DateTime.now()
        .add(Duration(seconds: expiresInSec))
        .millisecondsSinceEpoch;
    await _secure.write(key: _kExp, value: '$exp');
    if (refreshToken != null && refreshToken.isNotEmpty) {
      await _secure.write(key: _kRefresh, value: refreshToken);
    }
  }

  static Future<int?> segundosRestantes() async {
    final expStr = await _secure.read(key: _kExp);
    if (expStr == null || expStr.isEmpty) return null;
    final exp = int.tryParse(expStr);
    if (exp == null) return null;
    return ((exp - DateTime.now().millisecondsSinceEpoch) / 1000).floor();
  }

  static Future<String?> refreshToken() async {
    final t = await _secure.read(key: _kRefresh);
    if (t == null || t.isEmpty) return null;
    return t;
  }

  /// Token vigente o null (caduca 60 s antes).
  static Future<String?> tokenVigente() async {
    final t = await _secure.read(key: _kToken);
    if (t == null || t.isEmpty) return null;
    final expStr = await _secure.read(key: _kExp);
    final exp = int.tryParse(expStr ?? '') ?? 0;
    final limite = DateTime.now()
        .add(const Duration(seconds: 60))
        .millisecondsSinceEpoch;
    if (exp <= limite) return null;
    return t;
  }

  static Future<DateTime?> expiraEn() async {
    final s = await segundosRestantes();
    if (s == null) return null;
    return DateTime.now().add(Duration(seconds: s));
  }

  static Future<String?> leerRefresh() => refreshToken();

  static Future<void> clear() async {
    await _secure.delete(key: _kToken);
    await _secure.delete(key: _kExp);
    await _secure.delete(key: _kRefresh);
  }
}

/// Renueva el access token con el refresh de GoTrue. No imprime tokens.
class RenovarToken {
  static Future<bool>? _vuelo;

  static Future<bool> asegurar({bool forzar = false}) async {
    final enCurso = _vuelo;
    if (enCurso != null) return enCurso;
    if (!forzar) {
      final jwt = await SesionNube.tokenVigente();
      final exp = await SesionNube.expiraEn();
      final holgado =
          jwt != null &&
          jwt.isNotEmpty &&
          (exp == null ||
              exp.isAfter(DateTime.now().add(const Duration(minutes: 2))));
      if (holgado) return true;
      if (_vuelo != null) return _vuelo!;
    }
    final vuelo = _ejecutar();
    _vuelo = vuelo;
    try {
      return await vuelo;
    } finally {
      if (identical(_vuelo, vuelo)) _vuelo = null;
    }
  }

  static Future<bool> _ejecutar() async {
    final refresh = await SesionNube.leerRefresh();
    if (refresh == null || refresh.isEmpty) return false;
    final base = SupabaseConfig.url;
    if (base.isEmpty) return false;
    final anon = SupabaseConfig.anonKey;
    try {
      final res = await http
          .post(
            Uri.parse('$base/auth/v1/token?grant_type=refresh_token'),
            headers: {
              'apikey': anon,
              'Authorization': 'Bearer $anon',
              'Content-Type': 'application/json',
            },
            body: jsonEncode({'refresh_token': refresh}),
          )
          .timeout(const Duration(seconds: 4));
      if (res.statusCode == 400 ||
          res.statusCode == 401 ||
          res.statusCode == 403) {
        debugPrint('RenovarToken: rechazado ${res.statusCode}');
        await SesionNube.clear();
        return false;
      }
      if (res.statusCode != 200) {
        debugPrint('RenovarToken: http ${res.statusCode}');
        return false;
      }
      final decoded = jsonDecode(res.body);
      if (decoded is! Map) return false;
      final access = decoded['access_token']?.toString() ?? '';
      final nuevo = decoded['refresh_token']?.toString() ?? '';
      final raw = decoded['expires_in'];
      final secs = raw is int
          ? raw
          : raw is num
              ? raw.toInt()
              : int.tryParse('$raw') ?? 3600;
      if (access.isEmpty) return false;
      await SesionNube.guardar(
        token: access,
        expiresInSec: secs,
        refreshToken: nuevo.isEmpty ? null : nuevo,
      );
      debugPrint('RenovarToken: renovado (${secs}s)');
      return true;
    } catch (e) {
      debugPrint('RenovarToken: red $e');
      return false;
    }
