     7	class CampanaOfertaNotifier {
     8	  CampanaOfertaNotifier._();
     9	
    10	  static final AvisosRemote _api = AvisosRemote();
    11	
    12	  /// Las 'nuevas_cuentas' aun no existen como negocio al crearse la campana;
    13	  /// su descubrimiento es el selector del alta (setup_cuenta_page).
    14	  static bool _debeAvisar(String segmento) => segmento != 'nuevas_cuentas';
    15	
    16	  static Future<void> anunciarCreacion({
    17	    required String nombre,
    18	    required String descripcion,
    19	    required String tipo,
    20	    required String valor,
    21	    required String segmento,
    22	    String? planAplicar,
    23	  }) async {
    24	    if (!_debeAvisar(segmento)) return;
    25	    final esPlan = segmento == 'plan_especifico';
    26	    final cuerpo = descripcion.trim().isNotEmpty
    27	        ? descripcion.trim()
    28	        : 'Nueva oferta disponible (${_tipoTxt(tipo)}: $valor). '
    29	            'Entra en "Ofertas disponibles" para aprovecharla.';
    30	    await _enviar(
    31	      titulo: 'Nueva oferta: $nombre',
    32	      mensaje: cuerpo,
    33	      segmento: esPlan ? 'plan' : 'todos',
    34	      plan: esPlan ? planAplicar : null,
    35	    );
    36	  }
    37	
    38	  static Future<void> anunciarActivacion(CampanaPromocional c) async {
    39	    if (!c.activa || !_debeAvisar(c.segmentoObjetivo)) return;
    40	    final esPlan = c.segmentoObjetivo == 'plan_especifico';
    41	    final cuerpo = c.descripcion.isNotEmpty
    42	        ? c.descripcion
    43	        : 'Ya disponible (${c.tipoEtiqueta}: ${c.valor}). '
    44	            'Entra en "Ofertas disponibles" para aprovecharla.';
    45	    await _enviar(
    46	      titulo: 'Oferta activa: ${c.nombre}',
    47	      mensaje: cuerpo,
    48	      segmento: esPlan ? 'plan' : 'todos',
    49	      plan: esPlan ? c.planAplicar : null,
    50	    );
    51	  }
    52	
    53	  static Future<void> _enviar({
    54	    required String titulo,
    55	    required String mensaje,
    56	    required String segmento,
    57	    String? plan,
    58	  }) async {
    59	    try {
    60	      await _api.enviar(
    61	        tipo: 'oferta',
    62	        titulo: titulo,
    63	        mensaje: mensaje,
    64	        segmento: segmento,
    65	        plan: plan,
    66	        prioridad: 'normal',
    67	      );
    68	    } catch (_) {
    69	      // El aviso nunca debe bloquear crear/activar la campana.
    70	    }
    71	  }
    72	
    73	  static String _tipoTxt(String t) => switch (t) {
    74	        'descuento' => 'descuento',
    75	        'meses_gratis' => 'meses gratis',
    76	        'cuentas_gratis' => 'cuentas gratis',
    77	        _ => 'beneficio',
    78	      };
    79	}
